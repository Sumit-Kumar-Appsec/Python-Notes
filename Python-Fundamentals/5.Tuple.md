# TUPLE
- TUPLES CAN STORE MULTIPLE ITEM IN A SINGLE VERIABLE, AND LIKE STRING THESE ARE IMMUTABLE AND CANT BE CHANGE

```python
items = ("item1", "item2", "item3", "item4", "item5")
print(items)
print(type(items))         -- strings in tuple

('item1', 'item2', 'item3', 'item4', 'item5')
<class 'tuple'>
```

```python
items_number = (1, 2, 3, 4, 5)   -- numbers in tuple
print(items_number)
print(type(items_number))

(1, 2, 3, 4, 5)
<class 'tuple'>
```

```python
repeat = ("hello",) * 5      -- multiply
print(repeat)
print(type(repeat))

('hello', 'hello', 'hello', 'hello', 'hello')
<class 'tuple'>
```

```python
mixed_items = (1, "hello", 3.14, True)   -- mixed variables
print(mixed_items)
print(type(mixed_items))

(1, 'hello', 3.14, True)
<class 'tuple'>
```

```python
combined = items + items_number         -- combining two tuple
print(combined)
print(type(combined))

('item1', 'item2', 'item3', 'item4', 'item5', 1, 2, 3, 4, 5)
<class 'tuple'>
```

```python
items = ("item1", "item2", "item3", "item4", "item5")
```

```python
items = ("item1", "item2", "item3", "item4", "item5")

Forward indexing:
item1  item2  item3  item4  item5
  0      1      2      3      4
  
Backward / negative indexing:
item1  item2  item3  item4  item5
 -5     -4     -3     -2     -1
 

print(items[0])
item1

print(items[1])
item2

print(items[2])
item3

print(items[-1])
item5

print(items[-2])
item4

```

```python
items = ("item1", "item2", "item3", "item4", "item5")             -- [starts:-stops]) 

print(items[-3:-1])               -- start from -3 but stop before -1 means dont include -1
('item3', 'item4')

print(items[-3:])                 -- start from -3 and stop at the end and include all
('item3', 'item4', 'item5')

string = " I am the god"
print(string[0:4])     -- starts from space[0] and ends at a[3] before m[4]
 I a

print(string[-1])     -- start from last d[-1]
d

print(string[-1:])     -- start from last d[-1] and ends in last means only index[-1] included
d

print(string[:-1])    -- starts from [0] and end on the last index [0] 
 I am the go

print(string[0:4] + string[-1])       -- index[0:4][ I a] + index[-1][d]
 I ad
```
